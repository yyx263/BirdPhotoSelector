cmake_minimum_required(VERSION 3.20)
project(BirdPhotoSelectorNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

# 显式固定 LibRaw 0.22.3；版本由头文件和运行时再次核对，禁止自动采用系统的不同版本。
set(LIBRAW_ROOT "" CACHE PATH "Pinned LibRaw 0.22.3 SDK or installed source prefix")
find_path(LIBRAW_INCLUDE_DIR NAMES libraw/libraw.h HINTS "${LIBRAW_ROOT}" "${LIBRAW_ROOT}/include")
find_library(LIBRAW_LIBRARY NAMES raw libraw HINTS "${LIBRAW_ROOT}/lib")
if(NOT LIBRAW_INCLUDE_DIR OR NOT LIBRAW_LIBRARY)
  message(FATAL_ERROR "Build/install LibRaw 0.22.3 and set LIBRAW_ROOT to that exact prefix.")
endif()
add_library(BirdPhotoSelector.Raw SHARED raw-decoder.cpp)
target_include_directories(BirdPhotoSelector.Raw PRIVATE "${LIBRAW_INCLUDE_DIR}")
target_link_libraries(BirdPhotoSelector.Raw PRIVATE "${LIBRAW_LIBRARY}")
add_library(BirdPhotoSelector.Platform.Native SHARED platform.cpp)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(BirdPhotoSelector.Platform.Native PRIVATE platform-macos.mm)
  target_link_libraries(BirdPhotoSelector.Platform.Native PRIVATE "-framework Foundation")
  set_target_properties(BirdPhotoSelector.Raw PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(BirdPhotoSelector.Raw PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
foreach(target BirdPhotoSelector.Raw BirdPhotoSelector.Platform.Native)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()
install(TARGETS BirdPhotoSelector.Raw BirdPhotoSelector.Platform.Native LIBRARY DESTINATION . RUNTIME DESTINATION .)
